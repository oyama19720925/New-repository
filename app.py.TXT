def get_fundamental_data(code, current_price=None):
    """J-Quants /v2/fins/summary からファンダメンタルデータを取得・計算"""
    import requests

    API_KEY = '4Xl4a3FwEyaB1N3XHauyU3BOvFegzLKgWKBdAs0TqLU'

    if len(str(code)) == 4:
        code_str = str(code) + '0'
    else:
        code_str = str(code)

    try:
        r = requests.get(
            'https://api.jquants.com/v2/fins/summary',
            headers={'X-API-KEY': API_KEY},
            params={'code': code_str},
            timeout=10
        )
        if r.status_code != 200:
            return None

        items = r.json().get('data', [])
        if not items:
            return None

        latest = items[-1]

        def to_float(val, default=None):
            try:
                return float(val) if val not in ('', None) else default
            except:
                return default

        eps    = to_float(latest.get('EPS'))
        feps   = to_float(latest.get('FEPS'))
        eq     = to_float(latest.get('Eq'))
        ta     = to_float(latest.get('TA'))
        np_    = to_float(latest.get('NP'))
        sales  = to_float(latest.get('Sales'))
        op     = to_float(latest.get('OP'))
        eq_ar  = to_float(latest.get('EqAR'))
        fdiv   = to_float(latest.get('FDivAnn'))
        fop    = to_float(latest.get('FOP'))
        fsales = to_float(latest.get('FSales'))
        cfo    = to_float(latest.get('CFO'))
        sh_out = to_float(latest.get('ShOutFY'))
        tr_sh  = to_float(latest.get('TrShFY'), 0)

        bps = to_float(latest.get('BPS'))
        if bps is None and eq and sh_out:
            net_sh = sh_out - (tr_sh or 0)
            if net_sh > 0:
                bps = eq / net_sh

        roe = to_float(latest.get('ROE'))
        if roe is None and np_ and eq and eq > 0:
            roe = (np_ / eq) * 100

        per = pbr = div_yield = None
        if current_price and current_price > 0:
            use_eps = feps if feps else eps
            if use_eps and use_eps > 0:
                per = round(current_price / use_eps, 2)
            if bps and bps > 0:
                pbr = round(current_price / bps, 2)
            if fdiv and fdiv > 0:
                div_yield = round((fdiv / current_price) * 100, 2)

        return {
            'PER': per, 'PBR': pbr,
            'ROE': round(roe, 2) if roe else None,
            'EPS': eps, 'FEPS': feps,
            'BPS': round(bps, 1) if bps else None,
            'DivYield': div_yield, 'FDivAnn': fdiv,
            'Sales': sales, 'OP': op, 'NP': np_,
            'FSales': fsales, 'FOP': fop,
            'Eq': eq, 'TA': ta, 'EqAR': eq_ar, 'CFO': cfo,
            'DiscDate': latest.get('DiscDate'),
            'DocType':  latest.get('DocType'),
            'CurPerType': latest.get('CurPerType'),
            'CurFYEn':  latest.get('CurFYEn'),
        }

    except Exception as e:
        st.warning(f'ファンダメンタルデータ取得エラー: {e}')
        return None

@st.cache_data(ttl=1800)
def fetch_fundamental(code: str, current_price: float = None):
    """
    /v2/fins/summary から EPS, BPS, ROE 等を取得し
    PER, PBR を計算して返す
    """
    headers = {"X-API-KEY": API_KEY}
    try:
        url = "https://api.jquants.com/v2/fins/summary"
        r = requests.get(url, headers=headers,
                         params={"code": code}, timeout=15)
        if r.status_code != 200:
            return {"error": f"HTTP {r.status_code}: {r.text[:200]}"}

        data = r.json()
        items = data.get("fins_summary", data.get("summary", []))
        if not items:
            return {"error": "データなし"}

        latest = items[-1] if isinstance(items, list) else items

        # 各指標を取得
        eps  = latest.get("EPS")   or latest.get("eps")
        bps  = latest.get("BPS")   or latest.get("bps")
        roe  = latest.get("ROE")   or latest.get("roe")
        sales = latest.get("Sales") or latest.get("sales")
        op   = latest.get("OP")    or latest.get("op")
        np_  = latest.get("NP")    or latest.get("np")
        eq   = latest.get("Eq")    or latest.get("eq")
        cfo  = latest.get("CFO")   or latest.get("cfo")
        feps = latest.get("FEPS")  or latest.get("feps")
        fdiv = latest.get("FDivAnn") or latest.get("fdivann")

        # PER, PBR 計算
        per, pbr = None, None
        if current_price and eps and float(eps) != 0:
            per = round(float(current_price) / float(eps), 2)
        if current_price and bps and float(bps) != 0:
            pbr = round(float(current_price) / float(bps), 2)

        return {
            "PER":   per,
            "PBR":   pbr,
            "ROE":   roe,
            "EPS":   eps,
            "BPS":   bps,
            "売上高": sales,
            "営業利益": op,
            "純利益": np_,
            "純資産": eq,
            "営業CF": cfo,
            "予想EPS": feps,
            "予想配当": fdiv,
            "raw": latest
        }
    except Exception as e:
        return {"error": str(e)}

@st.cache_data
def load_data(path: str):
    df = None
    for enc in ['utf-8', 'utf-8-sig', 'cp932', 'shift-jis']:
        try:
            df = pd.read_csv(path, parse_dates=['Date'], encoding=enc)
            break
        except:
            continue
    if df is None:
        return pd.DataFrame()

    col = df.columns.tolist()
    rename_map = {}
    for name in ['MktNm', 'Market', 'market']:
        if name in col: rename_map[name] = 'Market'; break
    for name in ['CoName', 'Name', 'name', '銘柄名']:
        if name in col: rename_map[name] = 'Name'; break
    for name in ['S33Nm', 'Sector', 'sector', '業種']:
        if name in col: rename_map[name] = 'Sector'; break
    for name in ['O', 'Open', 'open']:
        if name in col: rename_map[name] = 'Open'; break
    for name in ['H', 'High', 'high']:
        if name in col: rename_map[name] = 'High'; break
    for name in ['L', 'Low', 'low']:
        if name in col: rename_map[name] = 'Low'; break
    for name in ['C', 'Close', 'close']:
        if name in col: rename_map[name] = 'Close'; break
    for name in ['Vo', 'Volume', 'volume', 'vol', 'Vol', 'VOLUME']:
        if name in col: rename_map[name] = 'Volume'; break

    df.rename(columns=rename_map, inplace=True)

    if 'Volume' not in df.columns:
        df['Volume'] = 0
    df['Code'] = df['Code'].astype(str)
    return df

def calc_indicators(df, params):
    d = df.copy().sort_values('Date').reset_index(drop=True)
    s, l = params['ma_short'], params['ma_long']
    d['MA_s'] = d['Close'].rolling(s).mean()
    d['MA_l'] = d['Close'].rolling(l).mean()

    rsi_p = params['rsi_period']
    delta = d['Close'].diff()
    gain  = delta.clip(lower=0).rolling(rsi_p).mean()
    loss  = (-delta.clip(upper=0)).rolling(rsi_p).mean()
    rs    = gain / loss.replace(0, np.nan)
    d['RSI'] = 100 - 100 / (1 + rs)

    k_p, d_p = params['stoch_k'], params['stoch_d']
    lo = d['Low'].rolling(k_p).min()
    hi = d['High'].rolling(k_p).max()
    d['%K'] = (d['Close'] - lo) / (hi - lo + 1e-9) * 100
    d['%D'] = d['%K'].rolling(d_p).mean()

    d['EMA12']       = d['Close'].ewm(span=12).mean()
    d['EMA26']       = d['Close'].ewm(span=26).mean()
    d['MACD']        = d['EMA12'] - d['EMA26']
    d['Signal_MACD'] = d['MACD'].ewm(span=9).mean()

    bb_p = params['bb_period']
    d['BB_mid'] = d['Close'].rolling(bb_p).mean()
    d['BB_std'] = d['Close'].rolling(bb_p).std()
    d['BB_up']  = d['BB_mid'] + 2 * d['BB_std']
    d['BB_lo']  = d['BB_mid'] - 2 * d['BB_std']
    d['BB_%B']  = (d['Close'] - d['BB_lo']) / \
                  (d['BB_up'] - d['BB_lo'] + 1e-9) * 100

    if 'Volume' in d.columns:
        d['Vol_MA'] = d['Volume'].rolling(20).mean()
    else:
        d['Vol_MA'] = 0
    return d

# スクリーニング判定
# =============================================
def screen_stock(d, params, flags):
    if len(d) < 5:
        return False, {}
    last = d.iloc[-1]
    prev = d.iloc[-2]
    info = {}
    passed = True

    if flags.get('use_ma', False):
        gc    = (prev['MA_s'] <= prev['MA_l']) and (last['MA_s'] > last['MA_l'])
        ma_ok = gc if params['ma_cond'] == 'ゴールデンクロス' \
                else (last['MA_s'] > last['MA_l'])
        info['MA'] = '✅' if ma_ok else '❌'
        if not ma_ok: passed = False

    if flags.get('use_rsi', False):
        rsi_val = last['RSI']
        rsi_ok  = params['rsi_lo'] <= rsi_val <= params['rsi_hi']
        info['RSI'] = f"{'✅' if rsi_ok else '❌'} {rsi_val:.1f}"
        if not rsi_ok: passed = False

    if flags.get('use_stoch', False):
        k_val = last['%K']; d_val = last['%D']
        pk    = prev['%K']; pd_   = prev['%D']
        cond  = params['stoch_cond']
        if   cond == '%Kが%Dを上抜け（GC）':
            stoch_ok = (pk <= pd_) and (k_val > d_val)
        elif cond == '%K < 20（売られすぎ）':
            stoch_ok = k_val < 20
        elif cond == '%K > 80（買われすぎ）':
            stoch_ok = k_val > 80
        else:
            stoch_ok = params['stoch_lo'] <= k_val <= params['stoch_hi']
        info['Stoch'] = f"{'✅' if stoch_ok else '❌'} K:{k_val:.1f} D:{d_val:.1f}"
        if not stoch_ok: passed = False

    if flags.get('use_macd', False):
        macd_gc = (prev['MACD'] <= prev['Signal_MACD']) and \
                  (last['MACD'] > last['Signal_MACD'])
        info['MACD'] = '✅ GC' if macd_gc else '❌'
        if not macd_gc: passed = False

    if flags.get('use_bb', False):
        bb_ok = last['BB_%B'] <= params['bb_threshold']
        info['BB'] = f"{'✅' if bb_ok else '❌'} %B:{last['BB_%B']:.1f}"
        if not bb_ok: passed = False

    if flags.get('use_vol', False):
        vol_ok = last['Volume'] >= last['Vol_MA'] * params['vol_ratio']
        info['Vol'] = '✅' if vol_ok else '❌'
        if not vol_ok: passed = False

    info['Close']  = last['Close']
    info['Change'] = (last['Close'] - prev['Close']) / prev['Close'] * 100 \
                     if prev['Close'] != 0 else 0
    return passed, info

def run_backtest(df_stock, short_ma=5, long_ma=25,
                 initial_capital=1_000_000):
    df = df_stock.copy().sort_values('Date').reset_index(drop=True)
    if len(df) < long_ma + 5:
        return None, None

    df['MA_s'] = df['Close'].rolling(short_ma).mean()
    df['MA_l'] = df['Close'].rolling(long_ma).mean()

    trades, position = [], 0
    buy_price, buy_date, shares, capital = 0, None, 0, initial_capital
    equity, equity_dates = [initial_capital], [df.loc[0, 'Date']]

    for i in range(1, len(df)):
        ps = df.loc[i-1, 'MA_s']; pl = df.loc[i-1, 'MA_l']
        cs = df.loc[i,   'MA_s']; cl = df.loc[i,   'MA_l']
        if pd.isna(ps) or pd.isna(pl):
            equity.append(capital + position * shares * df.loc[i, 'Close'])
            equity_dates.append(df.loc[i, 'Date'])
            continue
        if ps <= pl and cs > cl and position == 0:
            shares     = max(100, int(capital // df.loc[i, 'Close'] // 100) * 100)
            buy_price  = df.loc[i, 'Close']
            buy_date   = df.loc[i, 'Date']
            position   = 1
            capital   -= shares * buy_price
        elif ps >= pl and cs < cl and position == 1:
            sp  = df.loc[i, 'Close']
            pnl = (sp - buy_price) * shares
            capital += shares * sp
            trades.append({
                '買日': buy_date, '売日': df.loc[i, 'Date'],
                '買値': buy_price, '売値': sp,
                '株数': shares,
                '損益(円)': round(pnl),
                '損益率(%)': round((sp - buy_price) / buy_price * 100, 2)
            })
            position = 0
        current = capital + position * shares * df.loc[i, 'Close']
        equity.append(current)
        equity_dates.append(df.loc[i, 'Date'])

    trades_df = pd.DataFrame(trades)
    equity_df = pd.DataFrame({'Date': equity_dates, 'Equity': equity})
    return trades_df, equity_df

def calc_summary(trades_df, equity_df, initial_capital=1_000_000):
    if trades_df is None or len(trades_df) == 0:
        return {}
    wins  = trades_df[trades_df['損益(円)'] > 0]
    loses = trades_df[trades_df['損益(円)'] <= 0]
    eq    = equity_df['Equity']
    dd    = ((eq - eq.cummax()) / eq.cummax() * 100).min()
    trades_df = trades_df.copy()
    trades_df['保有日数'] = (
        pd.to_datetime(trades_df['売日']) -
        pd.to_datetime(trades_df['買日'])
    ).dt.days
    return {
        '総損益(円)':   round(trades_df['損益(円)'].sum()),
        '勝率(%)':      round(len(wins)/len(trades_df)*100, 1),
        '総取引数':     len(trades_df),
        '勝':           len(wins),
        '負':           len(loses),
        'PF':           round(abs(wins['損益(円)'].sum() /
                         loses['損益(円)'].sum()), 2)
                        if loses['損益(円)'].sum() != 0 else 999,
        '最大DD(%)':    round(dd, 2),
        '平均保有日数': round(trades_df['保有日数'].mean(), 1),
        '最終資産(円)': round(eq.iloc[-1]),
        'リターン(%)':  round((eq.iloc[-1] - initial_capital) /
                         initial_capital * 100, 2)
    }

def plot_stock_chart(df_s, code, name, color):
    """ローソク足 + MA + 出来高チャート"""
    df_s = df_s.sort_values('Date').copy()
    df_s['MA5']  = df_s['Close'].rolling(5).mean()
    df_s['MA25'] = df_s['Close'].rolling(25).mean()
    df_s['MA75'] = df_s['Close'].rolling(75).mean()

    has_vol = 'Volume' in df_s.columns and df_s['Volume'].sum() > 0

    rows       = 2 if has_vol else 1
    row_h      = [0.75, 0.25] if has_vol else [1.0]
    sub_titles = (f'{code} {name}', '出来高') if has_vol \
                 else (f'{code} {name}',)

    fig = make_subplots(
        rows=rows, cols=1,
        shared_xaxes=True,
        row_heights=row_h,
        subplot_titles=sub_titles,
        vertical_spacing=0.05
    )
    # ローソク足
    fig.add_trace(go.Candlestick(
        x=df_s['Date'],
        open=df_s['Open'], high=df_s['High'],
        low=df_s['Low'],   close=df_s['Close'],
        increasing_line_color='#ff4444',
        decreasing_line_color='#00aaff',
        name='OHLC'
    ), row=1, col=1)
    # MA
    for ma_col, ma_color, ma_name in [
        ('MA5',  '#ffff00',  'MA5'),
        ('MA25', '#ff8800',  'MA25'),
        ('MA75', '#00ff88',  'MA75'),
    ]:
        fig.add_trace(go.Scatter(
            x=df_s['Date'], y=df_s[ma_col],
            mode='lines',
            line=dict(color=ma_color, width=1),
            name=ma_name
        ), row=1, col=1)
    # 出来高
    if has_vol:
        colors_vol = [
            '#ff4444' if c >= o else '#00aaff'
            for c, o in zip(df_s['Close'], df_s['Open'])
        ]
        fig.add_trace(go.Bar(
            x=df_s['Date'], y=df_s['Volume'],
            marker_color=colors_vol, name='出来高'
        ), row=2, col=1)

    fig.update_layout(
        height=420,
        margin=dict(l=5, r=5, t=40, b=5),
        plot_bgcolor='#0e1117',
        paper_bgcolor='#0e1117',
        font=dict(color='white'),
        xaxis_rangeslider_visible=False,
        showlegend=True,
        legend=dict(
            orientation='h', y=1.08,
            font=dict(size=10)
        )
    )
    fig.update_xaxes(gridcolor='#333')
    fig.update_yaxes(gridcolor='#333')
    return fig

def plot_equity(equity_df, trades_df, name, color):
    fig = make_subplots(
        rows=2, cols=1, row_heights=[0.7, 0.3],
        shared_xaxes=True,
        subplot_titles=('資産推移', 'ドローダウン'),
        vertical_spacing=0.08
    )
    r, g, b = (int(color[1:3],16),
               int(color[3:5],16),
               int(color[5:7],16))
    fig.add_trace(go.Scatter(
        x=equity_df['Date'], y=equity_df['Equity'],
        mode='lines', line=dict(color=color, width=2),
        fill='tozeroy',
        fillcolor=f'rgba({r},{g},{b},0.1)',
        name='資産'
    ), row=1, col=1)

    if trades_df is not None and len(trades_df) > 0:
        for _, t in trades_df.iterrows():
            be = equity_df[equity_df['Date'] <= t['買日']]['Equity']
            se = equity_df[equity_df['Date'] <= t['売日']]['Equity']
            if len(be) > 0:
                fig.add_trace(go.Scatter(
                    x=[t['買日']], y=[be.iloc[-1]],
                    mode='markers',
                    marker=dict(symbol='triangle-up', size=9, color='lime'),
                    showlegend=False
                ), row=1, col=1)
            if len(se) > 0:
                mc = 'red' if t['損益(円)'] < 0 else 'deepskyblue'
                fig.add_trace(go.Scatter(
                    x=[t['売日']], y=[se.iloc[-1]],
                    mode='markers',
                    marker=dict(symbol='triangle-down', size=9, color=mc),
                    showlegend=False
                ), row=1, col=1)

    eq = equity_df['Equity']
    dd = (eq - eq.cummax()) / eq.cummax() * 100
    fig.add_trace(go.Scatter(
        x=equity_df['Date'], y=dd,
        mode='lines', line=dict(color='tomato', width=1),
        fill='tozeroy',
        fillcolor='rgba(255,99,71,0.2)', name='DD'
    ), row=2, col=1)

    fig.update_layout(
        title=f'{name}', height=360,
        margin=dict(l=5, r=5, t=50, b=5),
        showlegend=False,
        plot_bgcolor='#0e1117', paper_bgcolor='#0e1117',
        font=dict(color='white')
    )
    fig.update_xaxes(gridcolor='#333')
    fig.update_yaxes(gridcolor='#333')
    return fig

def plot_heatmap(trades_df, name):
    if trades_df is None or len(trades_df) == 0:
        return None
    df = trades_df.copy()
    df['売日'] = pd.to_datetime(df['売日'])
    df['年']   = df['売日'].dt.year
    df['月']   = df['売日'].dt.month
    pivot = df.pivot_table(
        index='年', columns='月',
        values='損益(円)', aggfunc='sum'
    ).fillna(0)
    mn = {1:'1月',2:'2月',3:'3月',4:'4月',5:'5月',6:'6月',
          7:'7月',8:'8月',9:'9月',10:'10月',11:'11月',12:'12月'}
    pivot.columns = [mn.get(c, c) for c in pivot.columns]
    fig = px.imshow(
        pivot, color_continuous_scale='RdYlGn',
        color_continuous_midpoint=0,
        text_auto=True, aspect='auto',
        title=f'{name} 月別損益'
    )
    fig.update_layout(
        height=180, margin=dict(l=5,r=5,t=40,b=5),
        plot_bgcolor='#0e1117', paper_bgcolor='#0e1117',
        font=dict(color='white'), coloraxis_showscale=False
    )
    return fig
