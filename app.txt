# バックテスト
# =============================================
def run_backtest(df: pd.DataFrame, short: int, long: int, capital: int):
    df = df.sort_values("Date").reset_index(drop=True)
    if len(df) < long + 1:
        return None, None

    df["MA_S"] = df["AdjC"].rolling(short).mean()
    df["MA_L"] = df["AdjC"].rolling(long).mean()

    trades = []
    equity = [capital]
    cash   = capital
    pos    = None

    for i in range(1, len(df)):
        prev, curr = df.iloc[i - 1], df.iloc[i]
        if pd.isna(prev["MA_S"]) or pd.isna(prev["MA_L"]):
            equity.append(cash)
            continue

        # ゴールデンクロス → 買い
        if prev["MA_S"] < prev["MA_L"] and curr["MA_S"] >= curr["MA_L"] and pos is None:
            shares = cash // curr["AdjC"]
            if shares > 0:
                pos = {"entry_date": curr["Date"], "entry_price": curr["AdjC"], "shares": shares}
                cash -= shares * curr["AdjC"]

        # デッドクロス → 売り
        elif prev["MA_S"] > prev["MA_L"] and curr["MA_S"] <= curr["MA_L"] and pos is not None:
            pnl = (curr["AdjC"] - pos["entry_price"]) * pos["shares"]
            cash += pos["shares"] * curr["AdjC"]
            trades.append({
                "買日":    pos["entry_date"].strftime("%Y-%m-%d"),
                "売日":    curr["Date"].strftime("%Y-%m-%d"),
                "買値":    pos["entry_price"],
                "売値":    curr["AdjC"],
                "株数":    pos["shares"],
                "損益(円)":  round(pnl),
                "損益率(%)": round((curr["AdjC"] / pos["entry_price"] - 1) * 100, 2)
            })
            pos = None

        val = cash + (pos["shares"] * curr["AdjC"] if pos else 0)
        equity.append(val)

    trades_df = pd.DataFrame(trades) if trades else pd.DataFrame()
    equity_df = pd.DataFrame({"Date": df["Date"].values, "Equity": equity})
    return trades_df, equity_df

# =============================================
# サマリー計算
# =============================================
def calc_summary(trades_df, equity_df, capital):
    if trades_df is None or len(trades_df) == 0:
        return {}
    wins   = trades_df[trades_df["損益(円)"] > 0]
    losses = trades_df[trades_df["損益(円)"] <= 0]
    total_pnl   = trades_df["損益(円)"].sum()
    win_rate    = len(wins) / len(trades_df) * 100
    avg_win     = wins["損益(円)"].mean()   if len(wins)   > 0 else 0
    avg_loss    = losses["損益(円)"].mean() if len(losses) > 0 else 0
    pf          = abs(wins["損益(円)"].sum() / losses["損益(円)"].sum()) if len(losses) > 0 else float("inf")
    final_eq    = equity_df["Equity"].iloc[-1] if equity_df is not None else capital
    total_ret   = (final_eq / capital - 1) * 100

    # 最大ドローダウン
    eq = equity_df["Equity"]
    roll_max = eq.cummax()
    dd       = (eq - roll_max) / roll_max * 100
    max_dd   = dd.min()

    return {
        "総損益(円)":     round(total_pnl),
        "総リターン(%)":  round(total_ret, 2),
        "勝率(%)":        round(win_rate, 1),
        "プロフィットF":  round(pf, 2),
        "平均利益(円)":   round(avg_win),
        "平均損失(円)":   round(avg_loss),
        "最大DD(%)":      round(max_dd, 2),
        "トレード数":     len(trades_df)
    }

# 損益曲線チャート
# =============================================
def plot_equity(equity_df, trades_df, name, color):
    fig = go.Figure()
    fig.add_trace(go.Scatter(
        x=equity_df["Date"], y=equity_df["Equity"],
        mode="lines", line=dict(color=color, width=2),
        name="資産推移", fill="tozeroy",
        fillcolor=color.replace(")", ",0.1)").replace("rgb", "rgba")
    ))
    if trades_df is not None and len(trades_df) > 0:
        wins   = trades_df[trades_df["損益(円)"] > 0]
        losses = trades_df[trades_df["損益(円)"] <= 0]
        for subset, marker_color, symbol in [
            (wins,   "#00ff88", "triangle-up"),
            (losses, "#ff4444", "triangle-down")
        ]:
            if len(subset) > 0:
                dates  = pd.to_datetime(subset["売日"])
                equities = [
                    equity_df.loc[equity_df["Date"] == d, "Equity"].values[0]
                    if d in equity_df["Date"].values else np.nan
                    for d in dates
                ]
                fig.add_trace(go.Scatter(
                    x=dates, y=equities, mode="markers",
                    marker=dict(color=marker_color, size=8, symbol=symbol),
                    name="勝ち" if marker_color == "#00ff88" else "負け"
                ))
    fig.update_layout(
        height=250, margin=dict(l=0, r=0, t=20, b=0),
        paper_bgcolor="#0e0e1a", plot_bgcolor="#0e0e1a",
        font=dict(color="white"), showlegend=False,
        xaxis=dict(gridcolor="#333"), yaxis=dict(gridcolor="#333")
    )
    return fig

# =============================================
# ヒートマップ
# =============================================
def plot_heatmap(trades_df, name):
    if trades_df is None or len(trades_df) == 0:
        return None
    df = trades_df.copy()
    df["売日"] = pd.to_datetime(df["売日"])
    df["Year"]  = df["売日"].dt.year
    df["Month"] = df["売日"].dt.month
    pivot = df.groupby(["Year", "Month"])["損益(円)"].sum().unstack(fill_value=0)
    fig = go.Figure(go.Heatmap(
        z=pivot.values,
        x=[f"{m}月" for m in pivot.columns],
        y=pivot.index.astype(str),
        colorscale=[[0, "#ff4444"], [0.5, "#333"], [1, "#00ff88"]],
        showscale=False
    ))
    fig.update_layout(
        height=150, margin=dict(l=0, r=0, t=20, b=0),
        paper_bgcolor="#0e0e1a", plot_bgcolor="#0e0e1a",
        font=dict(color="white")
    )
    return fig

def fetch_fins_summary(code: str) -> dict | None:
    url     = "https://api.jquants.com/v2/fins/summary"
    headers = {"x-api-key": API_KEY}
    params  = {"code": code}
    try:
        r = requests.get(url, headers=headers, params=params, timeout=10)
        if r.status_code == 200:
            items = r.json().get("summary", [])
            return items[-1] if items else None
    except Exception as e:
        st.warning(f"APIエラー: {e}")
    return None
