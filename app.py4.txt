# 🔑 J-Quants API認証クラス（v2対応）
# ============================================
class JQuantsClient:
    def __init__(self, api_key, email, password):
        self.api_key = api_key
        self.email = email
        self.password = password
        self.refresh_token = None
        self.id_token = None
        self.base_url = JQUANTS_BASE
    
    def get_refresh_token(self):
        """リフレッシュトークンを取得（v2エンドポイント）"""
        url = f"{self.base_url}/auth/refresh_token"
        headers = {
            "Content-Type": "application/json"
        }
        payload = {
            "mail": self.email,
            "password": self.password
        }
        
        try:
            print(f"🔑 リフレッシュトークン取得中...")
            print(f"  URL: {url}")
            print(f"  メール: {self.email}")
            
            response = requests.post(url, headers=headers, json=payload)
            print(f"  ステータスコード: {response.status_code}")
            
            if response.status_code == 200:
                data = response.json()
                self.refresh_token = data.get("refreshToken")
                print(f"✅ リフレッシュトークン取得成功")
                return self.refresh_token
            else:
                print(f"❌ リフレッシュトークン取得失敗: {response.text}")
                return None
                
        except requests.exceptions.RequestException as e:
            print(f"❌ 通信エラー: {e}")
            return None
    
    def get_id_token(self):
        """IDトークンを取得"""
        if not self.refresh_token:
            self.get_refresh_token()
        
        if not self.refresh_token:
            print("❌ リフレッシュトークンがありません")
            return None
        
        url = f"{self.base_url}/auth/id_token"
        headers = {
            "Content-Type": "application/json"
        }
        payload = {
            "refreshToken": self.refresh_token
        }
        
        try:
            print(f"🔑 IDトークン取得中...")
            response = requests.post(url, headers=headers, json=payload)
            print(f"  ステータスコード: {response.status_code}")
            
            if response.status_code == 200:
                data = response.json()
                self.id_token = data.get("idToken")
                print(f"✅ IDトークン取得成功")
                return self.id_token
            else:
                print(f"❌ IDトークン取得失敗: {response.text}")
                return None
                
        except requests.exceptions.RequestException as e:
            print(f"❌ 通信エラー: {e}")
            return None
    
    def get_fundamental(self, code):
        """ファンダメンタルデータを取得（v2エンドポイント）"""
        if not self.id_token:
            self.get_id_token()
        
        if not self.id_token:
            print(f"❌ 認証に失敗しました")
            return None
        
        url = f"{self.base_url}/fundamental"
        params = {
            "code": code
        }
        headers = {
            "Authorization": f"Bearer {self.id_token}"
        }
        
        try:
            print(f"📊 {code} のファンダメンタルデータ取得中...")
            response = requests.get(url, headers=headers, params=params)
            print(f"  ステータスコード: {response.status_code}")
            
            if response.status_code == 200:
                data = response.json()
                print(f"✅ {code} のファンダメンタルデータ取得成功")
                return data
            else:
                print(f"❌ {code} のファンダメンタルデータ取得失敗: {response.text}")
                return None
                
        except requests.exceptions.RequestException as e:
            print(f"❌ 通信エラー: {e}")
            return None

def generate_dummy_data():

    names = ['トヨタ', 'ソニー', 'ソフトバンク', '東京エレクトロン', '信越化学']

        close = np.maximum(close, 100)
        
        df = pd.DataFrame({

            'Name': name,
            'Open': close * (1 + np.random.randn(90) * 0.01),

            'Low': close * (1 - np.random.randn(90) * 0.02),
            'Close': close,
            'Volume': np.random.randint(100000, 1000000, 90),
            'Sector': ['製造業', '電気機器', '通信', '半導体', '化学'][i],
            'Market': '東証プライム'
        })
        dfs.append(df)
    
    all_df = pd.concat(dfs, ignore_index=True)
    return all_df

# 📈 チャート描画関数
# ============================================
def create_candlestick_chart(df, code):
    """ローソク足チャートを作成"""
    stock_df = df[df['Code'] == code].copy()
    
    fig = go.Figure(data=[go.Candlestick(
        x=stock_df['Date'],
        open=stock_df['Open'],
        high=stock_df['High'],
        low=stock_df['Low'],
        close=stock_df['Close'],
        name='株価'
    )])
    
    # 移動平均線を追加
    stock_df['MA5'] = stock_df['Close'].rolling(window=5).mean()
    stock_df['MA25'] = stock_df['Close'].rolling(window=25).mean()
    
    fig.add_trace(go.Scatter(
        x=stock_df['Date'],
        y=stock_df['MA5'],
        mode='lines',
        name='MA5',
        line=dict(color='blue', width=1)
    ))
    
    fig.add_trace(go.Scatter(
        x=stock_df['Date'],
        y=stock_df['MA25'],
        mode='lines',
        name='MA25',
        line=dict(color='orange', width=1)
    ))
    
    fig.update_layout(
        title=f'{code} 株価チャート',
        xaxis_title='日付',
        yaxis_title='株価',
        xaxis_rangeslider_visible=False,
        template='plotly_white',
        height=600
    )
    
    return fig
